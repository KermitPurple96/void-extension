        API REFERENCE
        

          HTTP

          await void.request({url, method, headers, body}) — send request → {status, headers, body, elapsed}

          await void.history() — all captured HTTP history entries

          await void.cookies(url) — browser cookies as Map

          
Tabs

          void.sendToRepeater({url, method, headers, body}) — open in Repeater

          void.sendToIntruder({url, method, headers, body}) — load into Intruder

          await void.attack({url, method, headers, body, payloads, marker, injectIn, threads}) — run fuzzing attack. injectIn: "url"/"body"/"header". marker: string to replace (default "FUZZ"). Returns [{payload, status, length, elapsed, body, headers}]

          await void.scan({url, method, headers, body, modules}) — run scanner. modules: ["sqli","xss","pathtraversal","ssrf","ssti","cmdi","openredirect","headerinject"]

          
Encoding

          void.encode(value, ...ops) — chain encode: "base64", "url", "html", "hex", "unicode", "js", "url-double"

          void.decode(value, ...ops) — chain decode: "base64", "url", "html", "hex", "unicode", "js", "jwt"

          await void.hash(value, algo) — "md5", "sha1", "sha256"

          
Utility

          void.log(msg) — print to console   void.addFinding({title, severity, detail}) — add to Notes

          await void.sleep(ms) — wait   await void.storage(type) — read storage

          void.setVar(name, val) / void.getVar(name) — persist variables   void.isInScope(url) — scope check

          void.esc(str) — HTML-escape   void.parseUrl(url) — parse to {host, path, params}
